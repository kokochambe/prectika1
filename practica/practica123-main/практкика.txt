IF OBJECT_ID('dbo.audit_log', 'U') IS NOT NULL DROP TABLE dbo.audit_log;
IF OBJECT_ID('dbo.work_reports', 'U') IS NOT NULL DROP TABLE dbo.work_reports;

IF OBJECT_ID('dbo.work_orders', 'U') IS NOT NULL DROP TABLE dbo.work_orders;

IF OBJECT_ID('dbo.users', 'U') IS NOT NULL DROP TABLE dbo.users;

-- 2. СОЗДАНИЕ ТАБЛИЦ
CREATE TABLE dbo.users (
    id INT IDENTITY(1,1) PRIMARY KEY,
    username NVARCHAR(50) UNIQUE NOT NULL,
    password_hash NVARCHAR(128) NOT NULL DEFAULT 'pbkdf2_sha256$placeholder',
    role NVARCHAR(20) NOT NULL CHECK (role IN ('admin','engineer','technician','storekeeper','trainee')),
    department NVARCHAR(50),
    phone NVARCHAR(20),
    is_active BIT DEFAULT 1,
    created_at DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE dbo.equipment (
    id INT IDENTITY(1,1) PRIMARY KEY,
    inv_number NVARCHAR(30) UNIQUE NOT NULL,
    name NVARCHAR(100) NOT NULL,
    type NVARCHAR(50) NOT NULL,
    workshop NVARCHAR(50) NOT NULL,
    status NVARCHAR(20) DEFAULT 'active' CHECK (status IN ('active','maintenance','out_of_service','decommissioned')),
    install_date DATE,
    warranty_until DATE,
    last_maintenance DATE,
    created_at DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE dbo.work_orders (
    id INT IDENTITY(1,1) PRIMARY KEY,
    equipment_id INT FOREIGN KEY REFERENCES dbo.equipment(id) ON DELETE CASCADE,
    order_type NVARCHAR(30) NOT NULL CHECK (order_type IN ('planned','emergency','modernization','inspection')),
    priority INT NOT NULL DEFAULT 3 CHECK (priority BETWEEN 1 AND 5),
    status NVARCHAR(20) DEFAULT 'new' CHECK (status IN ('new','assigned','in_progress','pending_parts','completed','rejected')),
    assigned_user_id INT FOREIGN KEY REFERENCES dbo.users(id) ON DELETE SET NULL,
    created_by INT FOREIGN KEY REFERENCES dbo.users(id),
    description NVARCHAR(MAX) DEFAULT 'Техническое обслуживание / устранение неисправности',
    created_at DATETIME2 DEFAULT SYSDATETIME(),
    started_at DATETIME2,
    completed_at DATETIME2
);

CREATE TABLE dbo.maintenance_schedules (
    id INT IDENTITY(1,1) PRIMARY KEY,
    equipment_id INT FOREIGN KEY REFERENCES dbo.equipment(id) ON DELETE CASCADE,
    frequency_days INT NOT NULL,
    next_due DATE NOT NULL,
    last_completed DATE,
    checklist NVARCHAR(MAX) DEFAULT 'Проверка масла, натяжение, очистка фильтров, смазка, замер вибрации',
    is_active BIT DEFAULT 1
);

CREATE TABLE dbo.spare_parts (
    id INT IDENTITY(1,1) PRIMARY KEY,
    sku NVARCHAR(30) UNIQUE NOT NULL,
    name NVARCHAR(100) NOT NULL,
    category NVARCHAR(50),
    unit_price DECIMAL(10,2) NOT NULL CHECK (unit_price >= 0),
    min_stock INT DEFAULT 5,
    current_stock INT DEFAULT 0,
    warehouse_location NVARCHAR(30),
    updated_at DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE dbo.part_requests (
    id INT IDENTITY(1,1) PRIMARY KEY,
    work_order_id INT FOREIGN KEY REFERENCES dbo.work_orders(id) ON DELETE CASCADE,
    spare_part_id INT FOREIGN KEY REFERENCES dbo.spare_parts(id) ON DELETE NO ACTION,
    quantity INT NOT NULL CHECK (quantity > 0),
    status NVARCHAR(20) DEFAULT 'requested' CHECK (status IN ('requested','approved','issued','rejected')),
    requested_by INT FOREIGN KEY REFERENCES dbo.users(id),
    fulfilled_by INT FOREIGN KEY REFERENCES dbo.users(id),
    fulfilled_at DATETIME2
);

CREATE TABLE dbo.work_reports (
    id INT IDENTITY(1,1) PRIMARY KEY,
    work_order_id INT FOREIGN KEY REFERENCES dbo.work_orders(id) ON DELETE CASCADE,
    report_date DATE NOT NULL,
    hours_spent DECIMAL(4,1) NOT NULL,
    description NVARCHAR(MAX) DEFAULT 'Работы выполнены по регламенту. Замечаний нет.',
    technician_signature NVARCHAR(50),
    created_at DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE dbo.audit_log (
    id INT IDENTITY(1,1) PRIMARY KEY,
    user_id INT FOREIGN KEY REFERENCES dbo.users(id),
    action NVARCHAR(50) NOT NULL,
    table_name NVARCHAR(50) NOT NULL,
    record_id INT,
    old_value NVARCHAR(MAX),
    new_value NVARCHAR(MAX),
    timestamp DATETIME2 DEFAULT SYSDATETIME()
);

-- Индексы
CREATE INDEX idx_equipment_status ON dbo.equipment(status);
CREATE INDEX idx_work_orders_status ON dbo.work_orders(status);
CREATE INDEX idx_work_orders_assigned ON dbo.work_orders(assigned_user_id);
CREATE INDEX idx_part_requests_status ON dbo.part_requests(status);
CREATE INDEX idx_audit_log_timestamp ON dbo.audit_log(timestamp);
